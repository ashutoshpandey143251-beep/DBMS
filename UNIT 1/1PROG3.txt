DECLARE
    PNAME VARCHAR2(50);
    QTY NUMBER;
    PRICE NUMBER;
    DISCOUNT_PERCENT NUMBER;
    TOTAL_AMOUNT NUMBER;
    DISCOUNT_RS NUMBER;
BEGIN
    PNAME := '&PNAME';
    QTY := &QTY;
    PRICE := &PRICE;
    DISCOUNT_PERCENT := &DISCOUNT_PERCENT;

    TOTAL_AMOUNT := QTY * PRICE;
    DISCOUNT_RS := (TOTAL_AMOUNT * DISCOUNT_PERCENT) / 100;

    DBMS_OUTPUT.PUT_LINE('Product Name = ' || PNAME);
    DBMS_OUTPUT.PUT_LINE('Quantity = ' || QTY);
    DBMS_OUTPUT.PUT_LINE('Price = ' || PRICE);
    DBMS_OUTPUT.PUT_LINE('Total Amount = ' || TOTAL_AMOUNT);
    DBMS_OUTPUT.PUT_LINE('Discount = Rs. ' || DISCOUNT_RS);
END;
/
