DECLARE
    FEET NUMBER;
    CM NUMBER;
    INCH NUMBER;
    METER NUMBER;
BEGIN
    FEET := &FEET;

    CM := FEET * 30.48;
    INCH := FEET * 12;
    METER := FEET * 0.3048;

    DBMS_OUTPUT.PUT_LINE('Feet = ' || FEET);
    DBMS_OUTPUT.PUT_LINE('Centimeter = ' || CM);
    DBMS_OUTPUT.PUT_LINE('Inch = ' || INCH);
    DBMS_OUTPUT.PUT_LINE('Meter = ' || METER);
END;
/
