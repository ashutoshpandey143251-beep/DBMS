SET SERVEROUTPUT ON;

DECLARE
    v_name EMPLOYEE.ENAME%TYPE;
BEGIN
    SELECT ENAME
    INTO v_name
    FROM EMPLOYEE
    WHERE EID = 999;

    DBMS_OUTPUT.PUT_LINE('Employee Name: ' || v_name);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No employee found for the given ID.');

END;
/
